cmake_minimum_required(VERSION 3.20)
project(MYCODEXCoreProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(PREPEND CMAKE_MODULE_PATH "${OPENVPN3_ROOT}/cmake")
include(findcoredeps)
add_executable(mycodex_core_probe main.cpp "${MYCODEX_SDK_ROOT}/Sources/COpenVPNBridge/MYCODEXBridge.cpp")
target_include_directories(mycodex_core_probe PRIVATE "${MYCODEX_SDK_ROOT}/Sources/COpenVPNBridge/include")
target_compile_definitions(mycodex_core_probe PRIVATE MYCODEX_WITH_OPENVPN3 OPENVPN_FORCE_TUN_NULL)
add_core_dependencies(mycodex_core_probe)
